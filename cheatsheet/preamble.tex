% --- Headings ----------------------------------------------------------
% Purely typographic: centring, weight and size carry the hierarchy, so the
% page stays legible in greyscale and on a photocopier. Three distinct
% sizes, so a subsection never reads as a bold run of body text. One family
% throughout — the body roman, headings the same face in bold.
\renewcommand{\section}[1]{%
  \par\vspace{3pt}%
  {\centering\bfseries\Large #1\par}%
  \vspace{1pt}}

% Compact: these divide a chapter (Polarisation vs Diffraction), so they must
% read as a level above the bold run-in labels without costing a \large line.
% Centring is what outranks the left-aligned bold run-in labels, giving three
% levels: centred \Large section, centred bold subsection, left bold label.
% (No \scshape: Latin Modern has no bold small-caps shape and silently ignores
% it.) The \nobreak stops a heading being orphaned at the foot of a column.
\renewcommand{\subsection}[1]{%
  \par\vspace{2pt}%
  {\centering\bfseries #1\par}%
  \nobreak\vspace{0.5pt}\nobreak}

\renewcommand{\subsubsection}[1]{%
  \par\vspace{2pt}\noindent
  {\bfseries #1\par}}

% --- Shorthands --------------------------------------------------------
% Key result: centred on its own, no frame.
\newenvironment{key}
  {\par\vspace{2pt}\centering}
  {\par\vspace{2pt}}
\newcommand{\defn}[1]{\textbf{#1}}

% Diagrams: centred, \tiny labels, and far less vertical padding than center.
\newenvironment{fig}
  {\par\vspace{1pt}\centering\tiny}
  {\par\vspace{1pt}}
